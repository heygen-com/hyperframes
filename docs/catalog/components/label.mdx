## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `text` | `Workspace name` | string | The field name the label announces. |
| `badge` | `Required` | string | Wording inside the small chip that follows the field name. |
| `tone` | `muted` | `muted`, `accent`, `warning`, `danger` | Colour family of the chip: quiet grey, accent green, warning amber, or danger red. |
| `size` | `standard` | `compact`, `standard`, `roomy` | Label size, chip size, and the gap between them. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="label"
  data-composition-src="compositions/components/label.html"
  data-variable-values='{"text":"Workspace name","badge":"Required","tone":"muted","size":"standard"}'
></div>
```

      the warning and danger tokens.

<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    function text(value, fallback) {
      return typeof value === "string" && value.trim() !== "" ? value.trim() : fallback;
    }

    var tones = {
      muted: {
        color: "var(--_hf-ui-text-muted)",
        border: "var(--_hf-ui-border)",
      },
      accent: { color: "var(--_hf-ui-accent)", border: "var(--_hf-ui-accent)" },
      warning: {
        color: "var(--_hf-ui-warning)",
        border: "var(--_hf-ui-warning)",
      },
      danger: { color: "var(--_hf-ui-danger)", border: "var(--_hf-ui-danger)" },
    };
    var sizes = {
      compact: { size: "12px", gap: "6px", badge: "10px" },
      standard: { size: "13px", gap: "8px", badge: "11px" },
      roomy: { size: "15px", gap: "12px", badge: "13px" },
    };

    var tone = tones[pick(tones, vars.tone, "muted")];
    var size = sizes[pick(sizes, vars.size, "standard")];
    var labelText = text(vars.text, "Workspace name");
    var badgeText = text(vars.badge, "Required");

    var roots = document.querySelectorAll(".hf-ui-label");
    for (var i = 0; i < roots.length; i += 1) {
      var root = roots[i];
      root.style.setProperty("--hf-label-size", size.size);
      root.style.setProperty("--hf-label-gap", size.gap);
      root.style.setProperty("--hf-label-badge-size", size.badge);
      root.style.setProperty("--hf-label-badge-color", tone.color);
      root.style.setProperty("--hf-label-badge-border", tone.border);
      if (root.firstChild && root.firstChild.nodeType === 3) {
        root.firstChild.nodeValue = labelText + " ";
      }
      var badge = root.querySelector("span");
      if (badge) badge.textContent = badgeText;
    }
  })();
</script>
