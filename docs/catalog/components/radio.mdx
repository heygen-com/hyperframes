## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `accent` | `mint` | `mint`, `blue`, `violet` | Token family for the selected dot and the ring it sits in. |
| `size` | `standard` | `compact`, `standard`, `large` | Ring diameter; the dot, the gap and the label text scale with it. |
| `orientation` | `stacked` | `stacked`, `inline` | Stack the options down the frame or lay them out in a single row. |
| `options` | `Fast, Cinematic` | string | Comma separated option labels, one per radio, in order. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="radio"
  data-composition-src="compositions/components/radio.html"
  data-variable-values='{"accent":"mint","size":"standard","orientation":"stacked","options":"Fast, Cinematic"}'
></div>
```

  Animate --hf-radio-selected between 0 and 1 on each option from a paused
  GSAP timeline so renders stay deterministic.

  Variables. The script reads each one, falls back to the declared default on
  anything missing or unrecognised, and writes the result as a custom property

    - size (compact | standard | large, default standard): the ring diameter,
      16px / 20px / 26px. The dot, the gap and the label text scale with it.
    - orientation (stacked | inline, default stacked): stack the options down
      the frame, or lay them out in a single row.
    - options (comma separated text, default "Fast, Cinematic"): the label

<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    var accents = {
      mint: "var(--_hf-ui-accent)",
      blue: "var(--_hf-ui-data-2)",
      violet: "var(--_hf-ui-data-4)",
    };
    var sizes = { compact: "16px", standard: "20px", large: "26px" };
    var layouts = {
      stacked: { flow: "row", justify: "normal", gap: "12px" },
      inline: { flow: "column", justify: "start", gap: "24px" },
    };

    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    var accent = accents[pick(accents, vars.accent, "mint")];
    var size = sizes[pick(sizes, vars.size, "standard")];
    var layout = layouts[pick(layouts, vars.orientation, "stacked")];

    var labels = [];
    if (typeof vars.options === "string") {
      labels = vars.options.split(",").map(function (part) {
        return part.trim();
      });
    }
    if (labels.length === 0 || labels.join("") === "") {
      labels = ["Fast", "Cinematic"];
    }

    var groups = document.querySelectorAll(".hf-ui-radio-group");
    for (var g = 0; g < groups.length; g += 1) {
      groups[g].style.setProperty("--hf-radio-flow", layout.flow);
      groups[g].style.setProperty("--hf-radio-justify", layout.justify);
      groups[g].style.setProperty("--hf-radio-gap", layout.gap);

      var radios = groups[g].querySelectorAll(".hf-ui-radio");
      for (var i = 0; i < radios.length; i += 1) {
        radios[i].style.setProperty("--hf-radio-accent", accent);
        radios[i].style.setProperty("--hf-radio-size", size);
        if (typeof labels[i] !== "string" || labels[i] === "") continue;
        var node = radios[i].lastChild;
        while (node && node.nodeType !== 3) {
          node = node.previousSibling;
        }
        if (node) {
          node.nodeValue = labels[i];
        }
      }
    }
  })();
</script>
