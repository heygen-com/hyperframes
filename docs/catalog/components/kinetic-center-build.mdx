## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `size` | `standard` | `compact`, `standard`, `display` | Type size of the phrase. |
| `tone` | `ink` | `ink`, `paper`, `accent` | Word colour: ink for light frames, paper for dark ones, accent rides the brand token. |
| `spacing` | `standard` | `tight`, `standard`, `loose` | The gap between words in the phrase. |
| `direction` | `up` | `up`, `down`, `left`, `right` | Where the offset the timeline drives points, so words push in from below, above, or the side. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="kinetic-center-build"
  data-composition-src="compositions/components/kinetic-center-build.html"
  data-variable-values='{"size":"standard","tone":"ink","spacing":"standard","direction":"up"}'
></div>
```

<!--
  Kinetic Center Build - remocn typography primitive for HyperFrames.

  Paste the markup and CSS into a composition. Drive the timeline integration
  from a paused GSAP timeline so renders remain deterministic.

  Variables. The script reads each one, falls back to the declared default on
  anything missing or unrecognised, and writes a custom property the CSS above
  consumes. The timeline recipe below is untouched by them: it keeps driving
  one offset, and direction decides what that offset means on screen.

    - size (compact | standard | display, default standard): type size of the
      phrase, 40px, 56px, or 76px.
    - tone (ink | paper | accent, default ink): word colour. ink is near-black
      for light frames, paper near-white for dark ones, accent rides the
      brand token.
    - spacing (tight | standard | loose, default standard): the gap between
      words, 6px, 12px, or 22px.
    - direction (up | down | left | right, default up): where the offset the
      timeline drives points. up pushes the words into place from below, down
      from above, left and right along the baseline.

  On every default the result is identical to the original: near-black 56px
  words, 12px apart, pushing up into place.
-->

<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    var sizes = { compact: "40px", standard: "56px", display: "76px" };
    var tones = {
      ink: "#18181b",
      paper: "#fafafa",
      accent: "var(--brand, #34d399)",
    };
    var spacings = { tight: "6px", standard: "12px", loose: "22px" };
    var axes = {
      up: { x: 0, y: 1 },
      down: { x: 0, y: -1 },
      left: { x: -1, y: 0 },
      right: { x: 1, y: 0 },
    };

    var size = sizes[pick(sizes, vars.size, "standard")];
    var tone = tones[pick(tones, vars.tone, "ink")];
    var spacing = spacings[pick(spacings, vars.spacing, "standard")];
    var axis = axes[pick(axes, vars.direction, "up")];

    var roots = document.querySelectorAll(".hf-remocn-kinetic-center-build");
    for (var i = 0; i < roots.length; i += 1) {
      var root = roots[i];
      root.style.setProperty("--hf-word-size", size);
      root.style.setProperty("--hf-word-color", tone);
      root.style.setProperty("--hf-word-gap", spacing);
      root.style.setProperty("--hf-word-axis-x", String(axis.x));
      root.style.setProperty("--hf-word-axis-y", String(axis.y));
    }
  })();
</script>
