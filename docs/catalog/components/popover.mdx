## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `placement` | `below` | `below`, `above` | Which side of the anchor the panel sits on, and therefore which way it settles as it opens. |
| `gap` | `12` | 4px to 40px, step 1px | Distance between the anchor and the panel edge. |
| `accent` | `neutral` | `neutral`, `green`, `blue`, `violet` | Colour family for the panel outline and its heading. Neutral leaves the theme border and heading colours alone. |
| `label` | `Format` | string | Text on the button the panel is anchored to. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="popover"
  data-composition-src="compositions/components/popover.html"
  data-variable-values='{"placement":"below","gap":12,"accent":"neutral","label":"Format"}'
></div>
```

  Popover - anchored panel primitive.

  Animate --hf-popover-open from 0 to 1 on .hf-ui-popover.

  Variables. The script reads each one, falls back to the declared default on
  anything missing or invalid, and writes the result as a custom property the
  CSS below consumes. The timeline recipe stays untouched: it keeps driving one
  open value from 0 to 1, and these decide where the panel sits and how it
  looks while that value runs.

    - placement (below | above, default below): which side of the anchor the
      panel sits on. Below drops down into place, above rises up, and the
      transform origin follows so the panel still grows out of the button.
    - gap (number 4 to 40, default 12, px): distance between the anchor and the
      near edge of the panel. Values outside the range are clamped to it.
    - accent (neutral | green | blue | violet, default neutral): colour family
      for the panel outline and its heading. Neutral sets nothing, so a theme
      that restyles the shared border and text tokens is left alone.
    - label (string, default Format): text on the anchor button.

  On every default the result is identical to the original: a Format button
  with a neutral panel 12px below it.
-->

  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    var placements = {
      below: { slide: "-4px", origin: "top center", side: "top" },
      above: { slide: "4px", origin: "bottom center", side: "bottom" },
    };
    var accents = {
      neutral: null,
      green: "var(--_hf-ui-data-1)",
      blue: "var(--_hf-ui-data-2)",
      violet: "var(--_hf-ui-data-4)",
    };
    var defaultLabel = "Format";

    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    function clampNumber(value, min, max, fallback) {
      var n = typeof value === "string" ? Number(value.trim()) : value;
      if (typeof n !== "number" || !isFinite(n)) return fallback;
      return Math.min(max, Math.max(min, n));
    }

    var placement = placements[pick(placements, vars.placement, "below")];
    var accent = accents[pick(accents, vars.accent, "neutral")];
    var gap = clampNumber(vars.gap, 4, 40, 12);
    var label = typeof vars.label === "string" && vars.label.trim() ? vars.label.trim() : defaultLabel;
    var near = "calc(100% + " + gap + "px)";

    var roots = document.querySelectorAll(".hf-ui-popover");
    for (var i = 0; i < roots.length; i += 1) {
      var style = roots[i].style;
      style.setProperty("--hf-popover-slide", placement.slide);
      style.setProperty("--hf-popover-origin", placement.origin);
      style.setProperty("--hf-popover-top", placement.side === "top" ? near : "auto");
      style.setProperty("--hf-popover-bottom", placement.side === "bottom" ? near : "auto");
      if (accent) {
        style.setProperty("--hf-popover-border", accent);
        style.setProperty("--hf-popover-heading", accent);
      }

      var anchor = roots[i].querySelector(".hf-ui-popover-anchor");
      if (anchor) anchor.textContent = label;
    }
  })();
</script>
