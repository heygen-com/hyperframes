## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `rows` | `3` | 1 to 6, step 1 | Number of loading rows in the block. Values outside the range are clamped to it. |
| `density` | `standard` | `compact`, `standard`, `roomy` | Height of each row and the gap between them. |
| `fill` | `54` | 20% to 80%, step 1% | Width of the title bar inside each row, as a percent of the row. Values outside the range are clamped to it. |
| `slide` | `standard` | `subtle`, `standard`, `wide` | Scales the sideways travel the timeline drives, so rows settle from closer in or further out in the same time. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="skeleton-block"
  data-composition-src="compositions/components/skeleton-block.html"
  data-variable-values='{"rows":3,"density":"standard","fill":54,"slide":"standard"}'
></div>
```

  Skeleton Block - Operator Black solid loading rows.

  Paste the markup and CSS into a composition. Drive the timeline integration
  from a paused GSAP timeline so renders remain deterministic.

  Variables. The script reads each one, falls back to the declared default on
  anything missing or invalid, and writes the result as a custom property the
  CSS below consumes, or as a change to the row list itself. The timeline
  recipe stays untouched: it keeps driving one offset and one opacity, and
  slide decides how far that offset reaches.

    - rows (number 1 to 6, default 3): how many loading rows the block shows.
      Values outside the range are clamped to it.
    - density (compact | standard | roomy, default standard): height of each
      row and the gap between them, from a tight list to a generous one.
    - fill (number 20 to 80, default 54, %): width of the title bar inside each
      row, as a percent of the row. Values outside the range are clamped to it.
    - slide (subtle | standard | wide, default standard): scales the sideways
      travel the timeline drives, so rows settle from closer in or further out
      in the same time. At rest the offset is zero, so this changes nothing
      until the timeline runs.

  On every default the result is identical to the original: three 40px rows,
  12px apart, each with a 54 percent title bar, settling from 4px out.
-->

<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    var densities = {
      compact: { height: "30px", gap: "8px" },
      standard: null,
      roomy: { height: "52px", gap: "18px" },
    };
    var travels = { subtle: 0.5, standard: 1, wide: 3 };

    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    function clampNumber(value, min, max, fallback) {
      var n = typeof value === "string" ? Number(value.trim()) : value;
      if (typeof n !== "number" || !isFinite(n)) return fallback;
      return Math.min(max, Math.max(min, n));
    }

    var density = densities[pick(densities, vars.density, "standard")];
    var travel = travels[pick(travels, vars.slide, "standard")];
    var rows = Math.round(clampNumber(vars.rows, 1, 6, 3));
    var fill = Math.round(clampNumber(vars.fill, 20, 80, 54));

    var roots = document.querySelectorAll(".hf-remocn-skeleton-block");
    for (var i = 0; i < roots.length; i += 1) {
      var root = roots[i];
      var style = root.style;
      style.setProperty("--hf-block-fill", fill + "%");
      // The timeline drives --hf-row-x, so scale that offset rather than
      // replacing it: at rest it is zero and this stays invisible.
      style.setProperty("--hf-block-travel", String(travel));
      if (density) {
        style.setProperty("--hf-block-row-height", density.height);
        style.setProperty("--hf-block-gap", density.gap);
      }

      var list = root.querySelectorAll(".row");
      while (list.length > rows) {
        list[list.length - 1].remove();
        list = root.querySelectorAll(".row");
      }
      var template = list[list.length - 1];
      while (list.length < rows && template) {
        root.appendChild(template.cloneNode(true));
        list = root.querySelectorAll(".row");
      }
    }
  })();
</script>
