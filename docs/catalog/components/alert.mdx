## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `tone` | `info` | `info`, `success`, `warning`, `danger` | Status colour for the icon frame and glyph, and for the leading edge when it is enabled. |
| `edge` | `none` | `none`, `tone` | Leading edge treatment: none keeps the hairline border, tone thickens it into a status strip. |
| `width` | `standard` | `narrow`, `standard`, `wide` | How wide the alert sits. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="alert"
  data-composition-src="compositions/components/alert.html"
  data-variable-values='{"tone":"info","edge":"none","width":"standard"}'
></div>
```

<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    var tones = {
      info: { border: "var(--_hf-ui-control-border)", color: "var(--_hf-ui-text-muted)" },
      success: { border: "var(--_hf-ui-accent)", color: "var(--_hf-ui-accent)" },
      warning: { border: "var(--_hf-ui-warning)", color: "var(--_hf-ui-warning)" },
      danger: { border: "var(--_hf-ui-danger)", color: "var(--_hf-ui-danger)" },
    };
    var widths = { narrow: "320px", standard: "410px", wide: "560px" };

    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    var toneName = pick(tones, vars.tone, "info");
    var tone = tones[toneName];
    var edged = vars.edge === "tone";
    var width = widths[pick(widths, vars.width, "standard")];

    var roots = document.querySelectorAll(".hf-ui-alert");
    for (var i = 0; i < roots.length; i += 1) {
      var style = roots[i].style;
      roots[i].setAttribute("data-tone", toneName);
      style.setProperty("--hf-alert-tone-border", tone.border);
      style.setProperty("--hf-alert-tone-color", tone.color);
      style.setProperty("--hf-alert-edge-width", edged ? "3px" : "1px");
      style.setProperty(
        "--hf-alert-edge-color",
        edged ? tone.border : "var(--_hf-ui-border-strong)",
      );
      style.setProperty("--hf-alert-width", width);
    }
  })();
</script>
