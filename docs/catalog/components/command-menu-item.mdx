## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `labels` | `New composition,Render preview,Publish project` | string | Comma-separated row labels matched to rows by position. Blank entries keep the authored label. |
| `hover` | `2` | 1 to 8, step 1 | Which row carries the hover state, counting from 1. Out-of-range values clamp. |
| `accent` | `green` | `green`, `blue`, `violet` | Accent token behind the check icon and the focus ring. |
| `theme` | `dark` | `dark`, `light` | Token set for the rows: dark surfaces or light ones. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="command-menu-item"
  data-composition-src="compositions/components/command-menu-item.html"
  data-variable-values='{"labels":"New composition,Render preview,Publish project","hover":2,"accent":"green","theme":"dark"}'
></div>
```

<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.

    var accents = {
      green: "var(--_hf-ui-data-1)",
      blue: "var(--_hf-ui-data-2)",
      violet: "var(--_hf-ui-data-4)",
    };

    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    function list(value) {
      if (typeof value !== "string") return [];
      return value.split(",").map(function (part) {
        return part.trim();
      });
    }

    // 1-based row picker: anything unparseable or out of range clamps back in.
    function slot(value, count, fallback) {
      var n = typeof value === "number" ? value : parseInt(value, 10);
      if (!isFinite(n)) n = fallback;
      n = Math.round(n);
      if (n < 1) n = 1;
      if (n > count) n = count;
      return n - 1;
    }

    var accent = accents[pick(accents, vars.accent, "green")];
    var theme = vars.theme === "light" ? "light" : "dark";
    var labels = list(vars.labels);

    var roots = document.querySelectorAll(".hf-remocn-command-menu-item");
    for (var i = 0; i < roots.length; i += 1) {
      var root = roots[i];
      root.style.setProperty("--hf-ui-accent", accent);
      root.setAttribute("data-hf-theme", theme);

      var rows = root.querySelectorAll(".row");
      if (rows.length === 0) continue;

      var target = rows[slot(vars.hover, rows.length, 2)];
      var check = root.querySelector(".row svg");
      for (var j = 0; j < rows.length; j += 1) {
        var row = rows[j];
        if (row === target) {
          row.setAttribute("data-state", "hover");
        } else if (row.hasAttribute("data-state")) {
          row.setAttribute("data-state", "idle");
        }
        row.setAttribute("aria-selected", row === target ? "true" : "false");

        var labelEl = row.querySelector(".label");
        if (labelEl && labels[j]) labelEl.textContent = labels[j];
      }
      if (check) target.insertBefore(check, target.firstChild);
    }
  })();
</script>
