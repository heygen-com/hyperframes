## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `from-label` | `Before` | string | Caption on the outgoing scene, the one on screen when the transition starts. |
| `to-label` | `Zoom Through Transition` | string | Caption on the incoming scene, the one the timeline wipes into frame. |
| `depth` | `standard` | `close`, `standard`, `far` | Multiplies how far both scenes travel and how hard the outgoing one blurs, so the push covers a shorter or longer run in the same time. |
| `accent` | `blue` | `blue`, `green`, `violet` | Colour pair behind the incoming scene: the shipped indigo ramp, a --brand green, or an --accent-2 violet. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="zoom-through-transition"
  data-composition-src="compositions/components/zoom-through-transition.html"
  data-variable-values='{"from-label":"Before","to-label":"Zoom Through Transition","depth":"standard","accent":"blue"}'
></div>
```

<!--
  Zoom Through Transition - remocn component ported to HyperFrames authoring format.

  Paste the markup, CSS and script into a composition. Drive the timeline
  integration from a paused GSAP timeline so renders remain deterministic.

  Variables. The script reads each one, falls back to the declared default on
  anything missing or unrecognised, and writes the result either into the
  markup or into a custom property the CSS above consumes. The timeline recipe
  below is untouched by them: it keeps driving the same offsets, scales, blur
  and wipe, and these decide how far that push carries, what the two scenes
  say, and what colour the incoming scene is.

    - from-label (string, default Before): caption on the outgoing scene. It
      is the one on screen when the transition starts. Anything that is not a
      non-empty string keeps the shipped caption.
    - to-label (string, default Zoom Through Transition): caption on the
      incoming scene. It stays clipped out of frame until the timeline wipes
      it in. Anything that is not a non-empty string keeps the shipped
      caption.
    - depth (close | standard | far, default standard): multiplies how far
      both scenes travel and how hard the outgoing one blurs, so the push
      covers a shorter or longer run in the same time. close covers a little
      over half the shipped push, far a little under double it. Every input it
      multiplies is zero at rest, so it collapses to the shipped pose there.
      It deliberately leaves the two scale tracks alone: the timeline drives
      those from their own start values, which a multiplier would distort.
    - accent (blue | green | violet, default blue): colour pair behind the
      incoming scene. blue is the shipped indigo to violet ramp, green rides
      --brand into teal, violet rides --accent-2 into magenta. It only reads
      once the timeline has wiped the incoming scene into frame.

  On every default the result is identical to the original: Before pushing
  left and blurring while the indigo to violet scene wipes through.
-->

<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    var depths = { close: 0.55, standard: 1, far: 1.7 };
    // Each choice routes to a different contract token so the variable stays
    // meaningful under a theme rather than collapsing into one themed accent.
    var accents = {
      blue: { a: "#2563eb", b: "#7c3aed" },
      green: { a: "var(--brand, #34d399)", b: "#0f766e" },
      violet: { a: "var(--accent-2, #a78bfa)", b: "#db2777" },
    };

    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    function label(value) {
      return typeof value === "string" && value !== "" ? value : null;
    }

    var depth = depths[pick(depths, vars.depth, "standard")];
    var accent = accents[pick(accents, vars.accent, "blue")];
    var fromLabel = label(vars["from-label"]);
    var toLabel = label(vars["to-label"]);

    var roots = document.querySelectorAll(".hf-remocn-zoom-through-transition");
    for (var i = 0; i < roots.length; i += 1) {
      var root = roots[i];
      root.style.setProperty("--hf-depth", String(depth));
      root.style.setProperty("--hf-scene-a", accent.a);
      root.style.setProperty("--hf-scene-b", accent.b);

      var fromScene = root.querySelector(".from");
      var toScene = root.querySelector(".to");
      if (fromLabel !== null && fromScene) {
        fromScene.textContent = fromLabel;
      }
      if (toLabel !== null && toScene) {
        toScene.textContent = toLabel;
      }
    }
  })();
</script>
