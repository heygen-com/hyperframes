## Variables

Every one of these has a default, so the piece works untouched. Set the ones you
want to change on the element:

| Variable | Default | Accepts | What it does |
| --- | --- | --- | --- |
| `state` | `default` | `default`, `invalid`, `disabled` | Control state: default, invalid draws the danger border, disabled dims the control. |
| `width` | `standard` | `narrow`, `standard`, `wide` | How wide the field sits before it shrinks to the host box. |
| `density` | `comfortable` | `comfortable`, `compact` | Control height and type size. Compact tightens the field for stacked forms. |

Set them with `data-variable-values` on the element that mounts it. These are the
defaults, so this behaves exactly like the preview above until you change one:

```html wrap
<div
  data-composition-id="field"
  data-composition-src="compositions/components/field.html"
  data-variable-values='{"state":"default","width":"standard","density":"comfortable"}'
></div>
```

  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    var widths = { narrow: "260px", standard: "340px", wide: "460px" };
    var densities = {
      comfortable: { control: "40px", label: "13px", text: "14px", help: "12px" },
      compact: { control: "32px", label: "12px", text: "13px", help: "11px" },
    };
    var states = { default: "", invalid: "aria-invalid", disabled: "aria-disabled" };

    var width = widths[pick(widths, vars.width, "standard")];
    var density = densities[pick(densities, vars.density, "comfortable")];
    var stateAttr = states[pick(states, vars.state, "default")];

    var roots = document.querySelectorAll(".hf-ui-field");
    for (var i = 0; i < roots.length; i += 1) {
      var style = roots[i].style;
      style.setProperty("--hf-field-width", width);
      style.setProperty("--hf-ui-control-height", density.control);
      style.setProperty("--hf-field-label-size", density.label);
      style.setProperty("--hf-field-text-size", density.text);
      style.setProperty("--hf-field-help-size", density.help);
      var control = roots[i].querySelector(".hf-ui-field-control");
      if (control && stateAttr) {
        control.setAttribute(stateAttr, "true");
      }
    }
  })();
</script>
